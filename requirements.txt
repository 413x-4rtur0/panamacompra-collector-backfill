#   python3 -m venv .venv
#   . .venv/bin/activate
#   python -m pip install --upgrade pip
#   python -m pip install -r requirements.txt

# Runtime Python modules used by the collector:
#   - playwright: browser automation for index/detail downloads and day refreshes
#
# WAHA WhatsApp group notifications use Python's standard library (urllib/json)
# and do not require an additional pip dependency.
# The webhook health/defer workflow and monitor status flags also use only
# Python/shell standard tooling; no extra pip dependency is required.
#
# System packages are not installable by pip. On Debian/Ubuntu/Linux Mint install:
#   sudo apt install -y python3-venv python3-full python3-tk
# If pip fails with ModuleNotFoundError: _posixsubprocess, recreate .venv with a
# complete system Python after installing python3-venv/python3-full.
# Install Playwright's Firefox browser once for the active virtualenv:
#   python -m playwright install firefox
